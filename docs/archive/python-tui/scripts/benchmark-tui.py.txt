#!/usr/bin/env python3
"""Headless TUI measurements; not a substitute for Linux terminal release checks.

Run from the repository: .venv/bin/python scripts/benchmark-tui.py --samples 5
Each sample is a fresh process with a temporary HOME/workspace and a fake engine.
"""
from __future__ import annotations

import argparse
import asyncio
import importlib.metadata
import json
import math
import os
from pathlib import Path
import platform
import resource
import subprocess
import sys
import tempfile
import time

STARTED = time.perf_counter()
ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))


def percentile(values: list[float], fraction: float) -> float:
    return sorted(values)[max(0, math.ceil(len(values) * fraction) - 1)]


async def sample(idle_seconds: float) -> dict[str, float]:
    from textual.events import Key
    from textual.widgets import RichLog, TextArea

    from backend.app.core.cli import build_parser
    from backend.app.tui.app import LocalCodeApp

    options = build_parser().parse_args([
        "--engine", "fake", "--no-extensions", "--no-packages", "--in-memory",
        "--cwd", os.getcwd(),
    ])
    app = LocalCodeApp(options)
    result: dict[str, float] = {}
    async with app.run_test(size=(100, 32)):
        composer = app.query_one("#composer", TextArea)
        composer.focus()
        result["script_start_to_mounted_prompt_ms"] = (time.perf_counter() - STARTED) * 1000
        # Measure an actual key event, without Pilot.press's deliberate delays.
        before = time.perf_counter()
        composer.post_message(Key("x", "x"))
        async with asyncio.timeout(5):
            while composer.text != "x":
                await asyncio.sleep(0)
        result["key_to_editor_state_ms"] = (time.perf_counter() - before) * 1000
        painted = asyncio.Event()
        app.call_after_refresh(painted.set)
        await asyncio.wait_for(painted.wait(), 5)
        result["key_to_refresh_ms"] = (time.perf_counter() - before) * 1000

        sentinel = "BENCHMARK_FINAL_EVENT_7cf2"
        before = time.perf_counter()
        app.receive_frame({
            "type": "message_end",
            "message": {"role": "assistant", "content": [{"type": "text", "text": sentinel}]},
        })
        async with asyncio.timeout(5):
            while not any(sentinel in line.text for line in app.query_one("#transcript", RichLog).lines):
                await asyncio.sleep(0.001)
        painted = asyncio.Event()
        app.call_after_refresh(painted.set)
        await asyncio.wait_for(painted.wait(), 5)
        result["final_event_to_refresh_ms"] = (time.perf_counter() - before) * 1000
        # Let initialization, animation and disk activity settle before idle CPU.
        await asyncio.sleep(0.5)
        cpu, wall = time.process_time(), time.perf_counter()
        await asyncio.sleep(idle_seconds)
        result["idle_cpu_percent_one_core"] = (
            (time.process_time() - cpu) / (time.perf_counter() - wall) * 100
        )
        # ru_maxrss units differ across Linux/macOS. Includes core runtime thread,
        # excludes vendor children (none in this synthetic workload).
        rss = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
        result["peak_process_rss_mib"] = rss / (1024 ** 2 if sys.platform == "darwin" else 1024)
    return result


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--samples", type=int, default=5)
    parser.add_argument("--idle-seconds", type=float, default=5)
    parser.add_argument("--child", action="store_true", help=argparse.SUPPRESS)
    args = parser.parse_args()
    if args.samples < 1 or args.idle_seconds <= 0:
        parser.error("samples and idle-seconds must be positive")
    if args.child:
        print(json.dumps(asyncio.run(sample(args.idle_seconds))))
        return
    runs = []
    for _ in range(args.samples):
        with tempfile.TemporaryDirectory(prefix="localcode-benchmark-") as directory:
            env = {**os.environ, "HOME": directory, "TERM": "xterm-256color"}
            proc = subprocess.run(
                [sys.executable, str(Path(__file__).resolve()), "--child", "--idle-seconds", str(args.idle_seconds)],
                cwd=directory, env=env, capture_output=True, text=True,
                timeout=30 + args.idle_seconds,
            )
            if proc.returncode:
                sys.stderr.write(proc.stderr)
                raise SystemExit(proc.returncode)
            runs.append(json.loads(proc.stdout))
    print(json.dumps({
        "measurement": "source headless; fresh Python processes; OS filesystem cache uncontrolled",
        "platform": platform.platform(), "architecture": platform.machine(),
        "python": platform.python_version(), "textual": importlib.metadata.version("textual"),
        "terminal": "Textual headless 100x32 (no PTY, SSH, tmux or vendor children)",
        "sample_count": len(runs), "idle_seconds_per_sample": args.idle_seconds,
        "p95": {key: round(percentile([r[key] for r in runs], 0.95), 3) for key in runs[0]},
        "samples": runs,
    }, indent=2))


if __name__ == "__main__":
    main()
