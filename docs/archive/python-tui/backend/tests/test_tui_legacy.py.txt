from __future__ import annotations

import queue
import time
from pathlib import Path

from backend.app.orchestrator.base import Event
from backend.app.tui.runtime import Runtime
from backend.tests.test_tui_runtime import Events, options


class ScriptedProvider:
    name = "scripted"

    def __init__(self) -> None:
        self.contexts = []
        self.closed = []

    async def open_session(self, ctx):
        self.contexts.append(ctx)
        return ctx.upstream_session_id or "upstream-1"

    async def run(self, ctx):
        self.contexts.append(ctx)
        yield Event("assistant.text", {"text": f"answer: {ctx.prompt}"})
        yield Event("assistant.done", {"upstream_session_id": "upstream-1"})

    async def close_session(self, session_id):
        self.closed.append(session_id)

    async def aclose(self):
        pass


def test_legacy_metadata_messages_and_upstream_resume(project: Path, isolated_store: Path, monkeypatch) -> None:
    from backend.app.tui import legacy

    provider = ScriptedProvider()

    async def get_provider(name):
        assert name == "scripted"
        return provider

    monkeypatch.setattr(legacy, "get_provider", get_provider)
    events = Events()
    runtime = Runtime(options(project), events)
    runtime.start()
    events.until("ready")
    try:
        made = runtime.submit({"type": "legacy_new", "id": "n", "provider": "scripted",
                               "model": "m1", "cwd": str(project), "permission_mode": "plan",
                               "fleet_config_override": {"gate": True}}).result(5)
        assert made["success"]
        sid = made["data"]["id"]
        assert made["data"]["permission_mode"] == "plan"
        assert runtime.submit({"type": "legacy_prompt", "id": "p", "message": "first"}).result(5)["success"]
        done = events.until("legacy_event")
        while done["event"]["type"] != "assistant.done":
            done = events.until("legacy_event")
        msgs = runtime.submit({"type": "legacy_messages", "id": "m"}).result(5)["data"]["messages"]
        assert [m["role"] for m in msgs] == ["user", "assistant"]
        assert "answer: first" in str(msgs[1]["content"])
        assert runtime.submit({"type": "legacy_list", "id": "l"}).result(5)["data"]["sessions"][0]["id"] == sid
        opened = runtime.submit({"type": "legacy_open", "id": "o", "sessionId": sid}).result(5)
        assert opened["data"]["upstream_id"] == "upstream-1"
        assert runtime.submit({"type": "legacy_prompt", "id": "p2", "message": "second"}).result(5)["success"]
        done = events.until("legacy_event")
        while done["event"]["type"] != "assistant.done":
            done = events.until("legacy_event")
        assert provider.contexts[-1].upstream_session_id == "upstream-1"
        assert provider.contexts[-1].permission_mode == "plan"
        assert provider.contexts[-1].extras == {"fleet_config_override": {"gate": True}}
    finally:
        runtime.close().result(5)
