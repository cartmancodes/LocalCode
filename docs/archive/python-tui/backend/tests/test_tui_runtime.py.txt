from __future__ import annotations

import argparse
import queue
import time
from pathlib import Path

from backend.app.tui.runtime import Runtime


class Events:
    def __init__(self) -> None:
        self.items: queue.Queue[dict] = queue.Queue()

    def __call__(self, frame: dict) -> None:
        self.items.put(frame)

    def until(self, kind: str, timeout: float = 5) -> dict:
        end = time.monotonic() + timeout
        while time.monotonic() < end:
            item = self.items.get(timeout=max(0.01, end - time.monotonic()))
            if item.get("type") == kind:
                return item
        raise AssertionError(f"no {kind} event")


def options(project: Path, **overrides: object) -> argparse.Namespace:
    values = dict(engine="fake", model=None, cwd=str(project), session="new", thinking="off",
                  permission_mode=None, default_permission="ask", append_system_prompt=None,
                  trust_project=False, extension=[], no_extensions=True, in_memory=False,
                  models="", no_packages=True)
    values.update(overrides)
    return argparse.Namespace(**values)


def test_prompt_ack_stream_persistence_and_resume(project: Path, agent_dir: Path) -> None:
    events = Events()
    runtime = Runtime(options(project), events)
    runtime.start()
    ready = events.until("ready")
    assert ready["state"]["engine"] == "fake"
    try:
        assert runtime.submit({"type": "prompt", "message": "hello", "id": "1"}).result(5)["success"]
        events.until("agent_settled")
        state = runtime.submit({"type": "get_state", "id": "2"}).result(5)["data"]
        assert state["sessionFile"]
        assert runtime.submit({"type": "get_last_assistant_text", "id": "3"}).result(5)["data"] == {"text": "echo: hello"}
        path = state["sessionFile"]
    finally:
        runtime.close().result(5)

    resumed = Runtime(options(project, session=path), Events())
    resumed.start()
    try:
        result = resumed.submit({"type": "get_messages", "id": "4"}).result(5)
        assert [message["role"] for message in result["data"]["messages"]] == ["user", "assistant"]
    finally:
        resumed.close().result(5)


def test_dialog_response_during_prompt_and_abort_resolve_waiters(project: Path, agent_dir: Path) -> None:
    ext = project / "confirm.py"
    ext.write_text(
        "def setup(api):\n"
        "    async def approve(event, ctx):\n"
        "        if not await ctx.ui.confirm('Proceed?', 'tool'):\n"
        "            return {'block': True, 'reason': 'denied'}\n"
        "    api.on('tool_call', approve)\n"
    )
    events = Events()
    runtime = Runtime(options(project, extension=[str(ext)]), events)
    runtime.start()
    events.until("ready")
    try:
        assert runtime.submit({"type": "prompt", "message": "tool", "id": "p"}).result(5)["success"]
        # The fake engine needs a scripted tool; use a second controller test below for the
        # dialog path, while this asserts a concurrent command still gets through.
        assert runtime.submit({"type": "get_state", "id": "s"}).result(5)["success"]
        assert runtime.submit({"type": "abort", "id": "a"}).result(5)["success"]
    finally:
        runtime.close().result(5)


def test_errors_and_session_history_are_responses(project: Path, agent_dir: Path) -> None:
    events = Events()
    runtime = Runtime(options(project), events)
    runtime.start()
    events.until("ready")
    try:
        unknown = runtime.submit({"type": "missing", "id": "x"}).result(5)
        assert unknown["success"] is False and "unknown command" in unknown["error"]
        assert runtime.submit({"type": "prompt", "message": "history", "id": "p"}).result(5)["success"]
        events.until("agent_settled")
        rows = runtime.submit({"type": "list_sessions", "id": "l"}).result(5)["data"]["sessions"]
        assert rows
        session_path = runtime.submit({"type": "get_state", "id": "s"}).result(5)["data"]["sessionFile"]
        result = runtime.submit({"type": "delete_session", "sessionPath": session_path, "id": "d"}).result(5)
        assert result["success"] is False and "active" in result["error"]
    finally:
        runtime.close().result(5)
