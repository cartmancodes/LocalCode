from __future__ import annotations

import json
from pathlib import Path

import pytest

from backend.app.tui.commands import parse_command
from backend.app.tui.transcript import TranscriptStore, sanitize_terminal


def test_commands_reach_core_and_preserve_unknown_slashes() -> None:
    assert parse_command('/model fake/demo').payload == {'type': 'set_model', 'provider': 'fake', 'modelId': 'demo'}
    assert parse_command('/rpc {"type":"get_messages","id":"x"}').payload == {'type': 'get_messages', 'id': 'x'}
    assert parse_command('/some_extension hello').payload == {'type': 'prompt', 'message': '/some_extension hello'}
    assert parse_command('/legacy open abc').payload == {'type': 'legacy_open', 'sessionId': 'abc'}


def test_transcript_spools_complete_large_output_and_pages_bounded_preview(tmp_path: Path) -> None:
    store = TranscriptStore(tmp_path, window=2)
    huge = 'x' * 200_000
    for message in ('first', 'second', huge, 'fourth'):
        store.append('tool', message)
    recent = store.tail()
    assert [e.preview for e in recent] == ['x' * 2000 + '…', 'fourth']
    older = store.older(before=recent[0].index, limit=2)
    assert [e.preview for e in older] == ['first', 'second']
    assert store.read_full(recent[0].index) == huge
    store.close()


def test_terminal_control_sequences_are_removed() -> None:
    assert sanitize_terminal('ok\x1b[31mred\x1b[0m\x1b]0;title\x07\r\x00\b!') == 'okred!'
