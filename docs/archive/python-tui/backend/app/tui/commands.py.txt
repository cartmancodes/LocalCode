"""Translate terminal slash commands to the existing RPC/controller protocol."""

from __future__ import annotations

import json
import shlex
from dataclasses import dataclass
from typing import Any


@dataclass(frozen=True)
class Command:
    payload: dict[str, Any] | None = None
    local: str | None = None
    args: tuple[str, ...] = ()
    destructive: bool = False


HELP = """Enter sends · Shift+Enter inserts a line · Ctrl+Enter sends · Ctrl+C cancels · Ctrl+D quits
/help /sessions /resume PATH /new /name NAME /tree /fork ID /compact [instructions]
/model PROVIDER/MODEL /models /thinking LEVEL /cycle-model /cycle-thinking
/steer TEXT /followup TEXT /quota /stats /state /messages /entries /last
/engine claude|codex|fake /workspace PATH /history /older /output INDEX [BYTE_OFFSET]
/export INDEX PATH /artifact PATH /bash COMMAND /abort /packages /commands
/legacy list|open ID|new PROVIDER [MODEL]|send TEXT|approve ID yes|no|cancel|messages
/rpc JSON sends any core RPC command, including images and extension commands."""


def parse_command(text: str, *, legacy: bool = False) -> Command:
    raw = text.strip()
    if not raw:
        return Command(local="empty")
    if not raw.startswith("/"):
        return Command(payload={"type": "legacy_prompt" if legacy else "prompt", "message": text})
    name, _, rest = raw.partition(" ")
    name = name.lower()[1:]
    rest = rest.strip()
    try:
        parts = shlex.split(rest)
    except ValueError as exc:
        return Command(local="error", args=(str(exc),))
    if name == "rpc":
        try:
            payload = json.loads(rest)
            if not isinstance(payload, dict) or not isinstance(payload.get("type"), str):
                raise ValueError("/rpc expects a JSON object with a type")
            return Command(payload=payload)
        except (json.JSONDecodeError, ValueError) as exc:
            return Command(local="error", args=(str(exc),))
    if name in {"help", "quit", "exit", "history", "older", "dialogs"}:
        return Command(local="quit" if name == "exit" else name, args=tuple(parts))
    if name in {"output", "export", "artifact", "engine", "workspace"}:
        return Command(local=name, args=tuple(parts))
    simple = {
        "sessions": "list_sessions", "new": "new_session", "tree": "get_tree",
        "quota": "get_quota", "stats": "get_session_stats", "state": "get_state",
        "messages": "get_messages", "entries": "get_entries", "last": "get_last_assistant_text",
        "models": "get_available_models", "cycle-model": "cycle_model",
        "cycle-thinking": "cycle_thinking_level", "packages": "get_packages",
        "commands": "get_commands", "abort": "abort", "cancel": "abort",
        "queue-clear": "clear_queue", "clone": "clone", "fork-messages": "get_fork_messages",
        "thinking-levels": "get_available_thinking_levels", "auto-retry": "set_auto_retry",
        "abort-retry": "abort_retry", "legacy-list": "legacy_list",
    }
    if name in simple:
        return Command(payload={"type": simple[name]})
    if name in {"resume", "session"} and rest:
        return Command(payload={"type": "switch_session", "sessionPath": rest})
    if name == "name" and rest:
        return Command(payload={"type": "set_session_name", "name": rest})
    if name == "fork" and rest:
        return Command(payload={"type": "fork", "entryId": rest})
    if name == "compact":
        return Command(payload={"type": "compact", "customInstructions": rest or None})
    if name == "model" and "/" in rest:
        provider, model_id = rest.split("/", 1)
        return Command(payload={"type": "set_model", "provider": provider, "modelId": model_id})
    if name == "thinking" and rest:
        return Command(payload={"type": "set_thinking_level", "level": rest})
    if name == "steer" and rest:
        return Command(payload={"type": "steer", "message": rest})
    if name in {"followup", "follow-up"} and rest:
        return Command(payload={"type": "follow_up", "message": rest})
    if name == "bash" and rest:
        return Command(payload={"type": "bash", "command": rest})
    if name == "steering-mode" and rest:
        return Command(payload={"type": "set_steering_mode", "mode": rest})
    if name == "follow-up-mode" and rest:
        return Command(payload={"type": "set_follow_up_mode", "mode": rest})
    if name in {"auto-compaction", "auto-retry"} and rest:
        return Command(payload={"type": "set_auto_compaction" if name == "auto-compaction" else "set_auto_retry", "enabled": rest.lower() in {"1", "on", "true", "yes"}})
    if name == "delete" and rest:
        return Command(payload={"type": "delete_session", "sessionPath": rest}, destructive=True)
    if name == "legacy":
        if not parts:
            return Command(local="error", args=("/legacy requires a subcommand",))
        sub = parts[0].lower()
        if sub == "list":
            return Command(payload={"type": "legacy_list"})
        if sub == "open" and len(parts) > 1:
            return Command(payload={"type": "legacy_open", "sessionId": parts[1]})
        if sub == "new":
            return Command(payload={"type": "legacy_new", "provider": parts[1] if len(parts) > 1 else "claude", "model": parts[2] if len(parts) > 2 else None})
        if sub == "send" and len(parts) > 1:
            return Command(payload={"type": "legacy_prompt", "message": rest.partition(" ")[2]})
        if sub == "approve" and len(parts) > 2:
            return Command(payload={"type": "legacy_approve", "approval_id": parts[1], "decision": parts[2]})
        if sub in {"cancel", "abort"}:
            return Command(payload={"type": "legacy_abort"})
        if sub == "messages":
            return Command(payload={"type": "legacy_messages"})
    # Extensions, skills and templates are submitted as ordinary prompts.
    return Command(payload={"type": "legacy_prompt" if legacy else "prompt", "message": text})
