"""Thread-owned controller for the terminal client.

The terminal never awaits provider or persistence work.  ``emit`` is called on
the runtime thread and is allowed to apply backpressure to a bounded UI queue.
"""

from __future__ import annotations

import asyncio
import concurrent.futures
import contextlib
import os
import threading
from collections.abc import Callable
from typing import Any

from ..core.agent_session import AgentSession, create_agent_session
from ..core.cli import _models
from ..core.engines import create_engine
from ..core.rpc.server import RpcServer, RpcUIBridge
from ..core.session_manager import SessionManager
from .legacy import LegacyAdapter


class TerminalUIBridge(RpcUIBridge):
    """RPC dialogs with an explicit dismissal frame for terminal modals."""

    async def dialog(self, method: str, payload: dict[str, Any]) -> dict[str, Any]:
        before = set(self._pending)
        try:
            return await super().dialog(method, payload)
        finally:
            for rid in set(self._pending) - before:
                self._send({"type": "extension_ui_cancelled", "id": rid})

    def cancel_pending(self) -> None:
        for rid, future in list(self._pending.items()):
            if not future.done():
                future.set_result({"cancelled": True})
            self._send({"type": "extension_ui_cancelled", "id": rid})


class Runtime:
    def __init__(self, options: Any, emit: Callable[[dict[str, Any]], None]) -> None:
        self.options = options
        self.emit = emit
        self._thread: threading.Thread | None = None
        self._loop: asyncio.AbstractEventLoop | None = None
        self._started = threading.Event()
        self._lock = threading.Lock()
        self._closing = False
        self._closed: concurrent.futures.Future[None] | None = None
        self._session: AgentSession | None = None
        self._rpc: RpcServer | None = None
        self._ui: TerminalUIBridge | None = None
        self._core_started = False
        self._legacy: LegacyAdapter | None = None
        self._responses: dict[Any, concurrent.futures.Future[dict[str, Any]]] = {}

    def start(self) -> None:
        with self._lock:
            if self._thread is not None:
                return
            self._thread = threading.Thread(target=self._run, name="localcode-runtime", daemon=True)
            self._thread.start()
        self._started.wait()

    def _run(self) -> None:
        loop = asyncio.new_event_loop()
        self._loop = loop
        asyncio.set_event_loop(loop)
        self._started.set()
        self.emit({"type": "ready", "state": {
            "engine": getattr(self.options, "engine", "claude"),
            "cwd": os.path.abspath(getattr(self.options, "cwd", os.getcwd())),
            "session": getattr(self.options, "session", "new"),
            "isStreaming": False,
        }})
        try:
            loop.run_forever()
        finally:
            pending = asyncio.all_tasks(loop)
            for task in pending:
                task.cancel()
            if pending:
                loop.run_until_complete(asyncio.gather(*pending, return_exceptions=True))
            loop.close()

    def submit(self, command: dict[str, Any]) -> concurrent.futures.Future[dict[str, Any]]:
        self.start()
        with self._lock:
            if self._closing:
                failed: concurrent.futures.Future[dict[str, Any]] = concurrent.futures.Future()
                failed.set_exception(RuntimeError("runtime is closed"))
                return failed
            assert self._loop is not None
            return asyncio.run_coroutine_threadsafe(self._dispatch(command), self._loop)

    async def _ensure_core(self) -> None:
        if self._session is not None:
            return
        o = self.options
        engine = create_engine(getattr(o, "engine", "claude"))
        ui = TerminalUIBridge(self.emit)
        session = await create_agent_session(
            engine=engine,
            cwd=os.path.abspath(getattr(o, "cwd", os.getcwd())),
            session=getattr(o, "session", "new"),
            in_memory=getattr(o, "in_memory", False),
            extension_paths=getattr(o, "extension", []),
            discover_extensions=not getattr(o, "no_extensions", False),
            use_packages=not getattr(o, "no_packages", False),
            project_trusted=getattr(o, "trust_project", False),
            ui_bridge=ui,
            mode="rpc",
            model=getattr(o, "model", None),
            thinking_level=getattr(o, "thinking", "off"),
            append_system_prompt=getattr(o, "append_system_prompt", None),
            permission_mode=getattr(o, "permission_mode", None),
            default_permission=getattr(o, "default_permission", "ask"),
        )
        self._session = session
        self._ui = ui
        self._rpc = RpcServer(session, self._send, models=_models(getattr(o, "models", "")), ui_bridge=ui)

    def _send(self, frame: dict[str, Any]) -> None:
        self.emit(frame)
        if frame.get("type") == "response" and frame.get("id") in self._responses:
            self._responses[frame["id"]].set_result(frame)

    async def _dispatch(self, command: dict[str, Any]) -> dict[str, Any]:
        kind = command.get("type")
        try:
            if not isinstance(kind, str):
                raise ValueError("command type must be a string")
            if kind.startswith("legacy_"):
                if self._legacy is None:
                    self._legacy = LegacyAdapter(self.emit)
                data = await self._legacy.command(command)
                return self._respond(command, data)
            if kind == "list_sessions":
                cwd = command.get("cwd") or getattr(self.options, "cwd", os.getcwd())
                return self._respond(command, {"sessions": SessionManager.list(cwd)})
            if kind == "delete_session":
                path = str(command.get("sessionPath", ""))
                if not path:
                    raise ValueError("sessionPath is required")
                if self._session and self._session.session_file and os.path.realpath(path) == os.path.realpath(self._session.session_file):
                    raise ValueError("cannot delete the active session")
                if not SessionManager.delete(path):
                    raise FileNotFoundError(path)
                return self._respond(command, {"deleted": True})
            await self._ensure_core()
            assert self._rpc is not None and self._session is not None
            if kind == "extension_ui_response":
                self._rpc.ui.resolve(command)
                return self._respond(command)
            if kind == "navigate_tree":
                data = await self._session.navigate_tree(str(command.get("entryId", "")), summarize=bool(command.get("summarize", True)), custom_summary=command.get("customSummary"))
                return self._respond(command, data)
            if kind == "abort" and self._ui:
                self._ui.cancel_pending()
            if kind in {"prompt", "bash", "set_model", "compact", "fork", "new_session", "switch_session", "cycle_model"} and not self._core_started:
                await self._session.start()
                self._core_started = True
            if command.get("id") is None:
                command = {**command, "id": id(command)}
            response: concurrent.futures.Future[dict[str, Any]] = concurrent.futures.Future()
            self._responses[command["id"]] = response
            try:
                await self._rpc.handle_command(command)
                return response.result()
            finally:
                self._responses.pop(command["id"], None)
        except Exception as exc:
            return self._respond(command, error=f"{type(exc).__name__}: {exc}")

    def _respond(self, command: dict[str, Any], data: Any = None, *, error: str | None = None) -> dict[str, Any]:
        frame: dict[str, Any] = {"type": "response", "command": command.get("type"), "success": error is None}
        if command.get("id") is not None:
            frame["id"] = command["id"]
        if error is not None:
            frame["error"] = error
        elif data is not None:
            frame["data"] = data
        self.emit(frame)
        return frame

    def close(self) -> concurrent.futures.Future[None]:
        self.start()
        with self._lock:
            if self._closed is not None:
                return self._closed
            self._closing = True
            self._closed = concurrent.futures.Future()
            assert self._loop is not None
            asyncio.run_coroutine_threadsafe(self._shutdown(), self._loop)
            return self._closed

    async def _shutdown(self) -> None:
        try:
            if self._ui:
                self._ui.cancel_pending()
            if self._legacy:
                await self._legacy.close()
            if self._rpc:
                await self._rpc.close()
            if self._session:
                if self._core_started:
                    await self._session.shutdown("quit")
                else:
                    await self._session.engine.close()
                tasks = list(self._session._tasks)
                if tasks:
                    await asyncio.gather(*tasks, return_exceptions=True)
            assert self._closed is not None
            self._closed.set_result(None)
        except BaseException as exc:
            assert self._closed is not None
            self._closed.set_exception(exc)
        finally:
            asyncio.get_running_loop().call_soon(asyncio.get_running_loop().stop)
